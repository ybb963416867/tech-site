- @Proxy 通常用与对系统API调用的Hook
- @Insert 常用于操作APP与library的类

```java
public static ActivityRecord sActivityRecord;

static {
   sActivityRecord=new ActivityRecord();
}

//mayCreateSuper 的含义：如果这个类中没有这个方法时则创建一个
@Insert("onCreate",mayCreateSuper = true)
//后面的scope声明这个类的子类或者只有他自己
@TargetClass("android.support.v7.app.AppCompatActivity",scope = Scope.All)
//Activity 的原有的方法
protected void onCreate(Bundle savedInstanceState){
   sActivityRecord.mOnCreateTime = System.currentTimeMillis();
   Origin.callVoid();
}

@Insert("onWindowFocusChanged",mayCreateSuper = true)
//后面的scope声明这个类的子类或者只有他自己
@TargetClass("android.support.v7.app.AppCompatActivity",scope = Scope.All)
//Activity 的原有的方法
protected void onWindowFocusChanged(boolean hasFocus){
     sActivityRecord.mOnWindowsFocusChangedTime = System.currentTimeMillis();
     LogUtils.i("onWindowsFocusChanged cost " + ( sActivityRecord.mOnWindowsFocusChangedTime - sActivityRecord.mOnCreateTime))
   Origin.callVoid();
}

//声明方法的名字
@Proxy("i")
//声明目标类的路径
@TargetClass("android.util.Log")
public static int i(String tag,String msg){
  msg=msg+"ActivityHooker"
   //返回原有方法的返回值
   return (int)Origin.call();
}


public  class  ActivityRecord{

public static  int mOnCreateTime;

public static  int mOnWindowsFocusChangedTime;

}
```
